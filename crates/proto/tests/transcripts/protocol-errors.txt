# A server that breaks the protocol ends the session.

# BYE nobody asked for.
@ tls
S: * PREAUTH [CAPABILITY IMAP4rev1] ready
= greeted preauth
> list
C: A0001 LIST "" "*"
S: * BYE Autologout; idle for too long
! Bye("Autologout; idle for too long")
? Closed

# An answer to a command that was not sent.
@ tls
S: * PREAUTH ready
= greeted preauth
> list
C: A0001 LIST "" "*"
S: A0002 OK done
! Unexpected("tagged response")
@ tls
S: * PREAUTH ready
= greeted preauth
S: A0001 OK done
! Unexpected("tagged response")

# A go-ahead when nothing waits for one.
@ tls
S: * PREAUTH ready
= greeted preauth
S: + more?
! Unexpected("continuation request")

# A second greeting.
@ tls
S: * PREAUTH ready
= greeted preauth
S: * PREAUTH again
! Unexpected("PREAUTH")

# A literal larger than the cap is refused when it is announced.
@ tls
S: * PREAUTH ready
= greeted preauth
S: * 1 FETCH (BODY[] {67108865}
! LiteralTooLarge { size: 67108865, limit: 67108864 }

# A tagged line that is no status.
@ tls
S: * PREAUTH ready
= greeted preauth
S: A0001 HELLO
! Malformed("tagged response")

# Lists nested too deeply.
@ tls
S: * PREAUTH ready
= greeted preauth
S: * 1 FETCH (FLAGS (((((((((((((((((((((((((((((((((x)))))))))))))))))))))))))))))))))
! TooDeep { limit: 32 }

# What the client has no use for is passed over; an alert is shown, whatever the status word.
@ tls
S: * PREAUTH ready
= greeted preauth
S: * NO [ALERT] Disk almost full
S: * BAD junk
S: * ENABLED
S: * 3 RECENT
S: * FLAGS (\Seen)
S: * LIST () "/" Stray
S: * SEARCH 1 2
S: * 1 STORE
S: * x
= alert Disk almost full

# Message data only belongs to an open mailbox (RFC 9051 section 7.4): anywhere else it ends the session.
@ tls
S: * PREAUTH ready
= greeted preauth
S: * 3 EXISTS
! Unexpected("EXISTS response")
? Closed
@ tls
S: * PREAUTH ready
= greeted preauth
S: * 2 EXPUNGE
! Unexpected("EXPUNGE response")
@ tls
S: * PREAUTH ready
= greeted preauth
S: * 1 FETCH (UID 7 FLAGS (\Seen))
! Unexpected("FETCH response")

# Also after a SELECT the server refused: no mailbox is open then.
@ tls
S: * PREAUTH ready
= greeted preauth
> select INBOX
C: A0001 SELECT "INBOX"
S: A0001 NO [NONEXISTENT] no such mailbox
= A0001 No [NONEXISTENT] no such mailbox
S: * 1 FETCH (UID 7 FLAGS (\Seen))
! Unexpected("FETCH response")

# While LOGOUT of an open mailbox is under way, the mailbox is still open.
@ tls
S: * PREAUTH ready
= greeted preauth
> select INBOX
C: A0001 SELECT "INBOX"
S: A0001 OK [READ-WRITE] done
= A0001 selected exists=0 recent=0 flags=- permanent=- uidvalidity=none uidnext=none unseen=none modseq=none read-only=false
> logout
C: A0002 LOGOUT
S: * 2 EXPUNGE
S: * BYE logging out
S: A0002 OK done
= expunged 2
= A0002 logged out
