# STARTTLS on port 587: no password before TLS, EHLO again after it with what the server lists then.
@ plain
S: 220 smtp.example.org ESMTP
= greeted
> starttls
! WrongSmtpState { command: "STARTTLS", state: Connected }
> tls
! WrongSmtpState { command: "STARTTLS", state: Connected }
> ehlo client.example.org
C: EHLO client.example.org
S: 250-smtp.example.org
S: 250-SIZE 1000
S: 250 STARTTLS
= EHLO hello SIZE STARTTLS
> starttls
C: STARTTLS
S: 220 2.0.0 Ready to start TLS
= STARTTLS go ahead
? StartingTls
> mail ada@example.org
! NotEncrypted
> quit
! WrongSmtpState { command: "QUIT", state: StartingTls }
> tls
C: EHLO client.example.org
? Connected
S: 250-smtp.example.org
S: 250-SIZE 2000
S: 250 AUTH PLAIN
= EHLO hello SIZE AUTH
? Ready
> starttls
! AlreadyEncrypted
> auth-plain ada@example.org secret
C: AUTH PLAIN AGFkYUBleGFtcGxlLm9yZwBzZWNyZXQ=
S: 235 OK
= AUTH signed in
> auth-plain ada@example.org secret
! AlreadySignedIn

# Bytes after the go-ahead came before encryption: anyone on the way could have put them there.
@ plain
S: 220 smtp.example.org ESMTP
= greeted
> ehlo client.example.org
C: EHLO client.example.org
S: 250-smtp.example.org
S: 250 STARTTLS
= EHLO hello STARTTLS
> starttls
C: STARTTLS
S: 220 go ahead
S: 250-AUTH PLAIN
! PlaintextAfterStartTls
? Closed
> tls
! Closed
