# The greeting is 220; 554 and 421 mean no service; anything else is not a greeting.
@ tls
S: 554 5.3.2 No service for you
! Bye("5.3.2 No service for you")
@ plain
S: 554 No service, see http://example.org/
! Bye("")
@ tls
S: 421 busy
! Bye("busy")
@ tls
S: 250 hello
! Malformed("greeting")

# 421 at any time: the server is closing the session.
@ tls
S: 220 smtp.example.org ESMTP
= greeted
> ehlo client.example.org
C: EHLO client.example.org
S: 421 4.3.2 Shutting down
! Bye("4.3.2 Shutting down")
? Closed
@ tls
S: 220 smtp.example.org ESMTP
= greeted
S: 421 4.4.2 Timeout
! Bye("4.4.2 Timeout")

# A reply nobody asked for.

S: 250 OK
! Unexpected("reply")
@ tls
S: 220 smtp.example.org ESMTP
S: 250 OK
! Unexpected("reply")
